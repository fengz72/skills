cmake_minimum_required(VERSION 3.16)
project(dsl_ge_onnx_om LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CANN "$ENV{ASCEND_HOME_PATH}")
set(GE_CXX11_ABI "" CACHE STRING "C++ standard-library ABI of target GE")
set(DSL_BACKEND "" CACHE STRING "Add specialization: 0=cannbot, 1=tilelang")
if(NOT CANN OR NOT "${GE_CXX11_ABI}" MATCHES "^[01]$" OR NOT "${DSL_BACKEND}" MATCHES "^[01]$")
  message(FATAL_ERROR "Source CANN and pass GE_CXX11_ABI and DSL_BACKEND")
endif()
include_directories("${CANN}/include" "${CANN}/include/external" "${CMAKE_SOURCE_DIR}")
link_directories("${CANN}/lib64")
add_compile_definitions(_GLIBCXX_USE_CXX11_ABI=${GE_CXX11_ABI})
set(OPP "${CMAKE_BINARY_DIR}/opp")
set(OPP_LIB "${OPP}/op_graph/lib/linux/${CMAKE_SYSTEM_PROCESSOR}")
file(MAKE_DIRECTORY "${OPP_LIB}" "${OPP}/framework/onnx" "${OPP}/op_graph/include")
configure_file(ops.h "${OPP}/op_graph/include/ops.h" COPYONLY)
add_library(cust_opapi SHARED custom_op.cc)
target_compile_definitions(cust_opapi PRIVATE DSL_BACKEND=${DSL_BACKEND})
target_compile_options(cust_opapi PRIVATE -Wall -Wextra)
target_link_libraries(cust_opapi PRIVATE -Wl,--whole-archive custom_op_registry_static -Wl,--no-whole-archive -Wl,--no-as-needed lowering register gert ascendcl acl_rt dl)
target_link_options(cust_opapi PRIVATE -Wl,-Bsymbolic-functions)
set_target_properties(cust_opapi PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${OPP_LIB}")
add_library(dsl_onnx_parser SHARED parser.cc)
target_link_libraries(dsl_onnx_parser PRIVATE register graph)
set_target_properties(dsl_onnx_parser PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${OPP}/framework/onnx")
add_executable(model_exec model_exec.cc)
target_compile_options(model_exec PRIVATE -Wall -Wextra -Werror)
target_link_libraries(model_exec PRIVATE ascendcl acl_rt)
