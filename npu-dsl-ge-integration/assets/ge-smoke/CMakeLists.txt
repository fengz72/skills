cmake_minimum_required(VERSION 3.16)
project(dsl_ge_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CANN "$ENV{ASCEND_HOME_PATH}")
if(NOT CANN)
  message(FATAL_ERROR "Source the target CANN environment first")
endif()
set(OPP_ROOT "${CMAKE_BINARY_DIR}/opp")
set(OPP_LIB "${OPP_ROOT}/op_graph/lib/linux/${CMAKE_SYSTEM_PROCESSOR}")
file(MAKE_DIRECTORY "${OPP_LIB}/artifacts" "${OPP_ROOT}/op_graph/include")
configure_file("${CMAKE_SOURCE_DIR}/cannbot-export/add_host.so" "${OPP_LIB}/artifacts/libcannbot_add.so" COPYONLY)
configure_file(ops.h "${OPP_ROOT}/op_graph/include/ops.h" COPYONLY)
set(GE_CXX11_ABI "" CACHE STRING "C++ standard-library ABI matching the target GE build")
if(NOT "${GE_CXX11_ABI}" MATCHES "^[01]$")
  message(FATAL_ERROR "Pass -DGE_CXX11_ABI matching the target GE build (0 or 1)")
endif()
add_compile_definitions(_GLIBCXX_USE_CXX11_ABI=${GE_CXX11_ABI})
include_directories("${CANN}/include" "${CANN}/include/graph" "${CANN}/include/ge" "${CANN}/include/external" "${CANN}/opp/built-in/op_proto/inc" "${CMAKE_SOURCE_DIR}")
link_directories("${CANN}/lib64")
add_library(cust_opapi SHARED launcher.cc custom_op.cc)
target_link_libraries(cust_opapi PRIVATE -Wl,--no-as-needed "${OPP_LIB}/artifacts/libcannbot_add.so" lowering ascendcl acl_rt)
set_target_properties(cust_opapi PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${OPP_LIB}" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/artifacts")
add_executable(smoke_runner main.cc)
target_link_libraries(smoke_runner PRIVATE cust_opapi graph ge_runner ge_compiler ascendcl acl_rt graph_base)
